# Internal site search results — duplicate, low-quality content. The
# /search page is also tagged noindex via meta robots, but we disallow
# crawling here to save crawl budget too.
Disallow: /search
Disallow: /search?
Disallow: /*?q=
Disallow: /404

# Faceted-filter URLs on category pages. They produce thin, near-
# duplicate variants of the canonical /c/<slug> page; we already mark
# them noindex client-side, but disallowing them in robots stops Google
# from spending crawl budget fetching them in the first place.
Disallow: /*?stock=
Disallow: /*?disc=
Disallow: /*?ai=
Disallow: /*?pop=
Disallow: /*?min=
Disallow: /*?max=
Disallow: /*?sort=
# Allow ?page= explicitly: paginated category URLs ARE canonicalised
# back to /c/<slug>?page=N (they have rel=next/prev too) so we want
# Google to crawl them.

# Block tracking-only query params that 3rd parties append (UTM,
# Google Click ID, Facebook Click ID) so Google doesn't treat them as
# distinct URLs.
Disallow: /*?utm_
Disallow: /*&utm_
Disallow: /*?gclid=
Disallow: /*?fbclid=
